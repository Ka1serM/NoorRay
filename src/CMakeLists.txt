#! A denoiser plugin registers itself in a static initializer that nothing references, so its archive links whole.
function(noorray_link_denoiser_plugins target)
    if(TARGET NoorRay.OpenDlssRr)
        target_link_libraries(${target} PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,NoorRay.OpenDlssRr>")
    endif()
endfunction()
if(NOORRAY_OPENDLSS_RR)
    add_subdirectory(NoorRay.OpenDlssRr)
endif()
#! DiffRenderer's CMake calls this for each of its targets.
function(diffrenderer_code_options target)
    target_compile_features(${target} PRIVATE cxx_std_23)
    noorray_code_options(${target})
endfunction()

    add_subdirectory(DiffRenderer.Tests)
