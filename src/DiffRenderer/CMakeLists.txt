file(GLOB_RECURSE diffrenderer_sources CONFIGURE_DEPENDS *.cpp)
if(NOORRAY_GAUSSIAN_TRACING)
    list(APPEND diffrenderer_sources "${PROJECT_SOURCE_DIR}/src/NoorRay/Spectral/Conductors.cpp" "${PROJECT_SOURCE_DIR}/src/NoorRay/Spectral/SpectralTables.cpp")
else()
    list(FILTER diffrenderer_sources EXCLUDE REGEX "/GaussianTracing/|/GaussianProxies\\.cpp$")
endif()
add_library(DiffRenderer STATIC ${diffrenderer_sources})
add_library(DiffRenderer::DiffRenderer ALIAS DiffRenderer)
diffrenderer_code_options(DiffRenderer)
if(NOORRAY_GAUSSIAN_TRACING)
    target_compile_definitions(DiffRenderer PUBLIC NOORRAY_GAUSSIAN_TRACING=1)
endif()
target_include_directories(DiffRenderer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
find_package(ZLIB REQUIRED)
target_link_libraries(DiffRenderer PUBLIC NoorRHI::NoorRHI PRIVATE stb xatlas ZLIB::ZLIB)

#! Every .slang file directly in <Group>/<Name>/Shaders is an entry file; the modules that they import are in Shaders/Modules, and the modules that every group shares are in Shaders.
set(XrayProjectionShaderIncludes "${CMAKE_CURRENT_SOURCE_DIR}/Algorithms/VolumeRendering/Shaders/Modules")
set(noorray_shader_includes "${PROJECT_SOURCE_DIR}/src/NoorRay/Shaders" "${PROJECT_SOURCE_DIR}/src")
set(GaussianTracingShaderIncludes ${noorray_shader_includes})
set(TrainingShaderIncludes ${noorray_shader_includes})
#! The entry files of Gaussian tracing only import the modules that trace.
set(GaussianTracingShaderOptions RAY_QUERIES)
file(GLOB shader_folders LIST_DIRECTORIES true "${CMAKE_CURRENT_SOURCE_DIR}/Algorithms/*" "${CMAKE_CURRENT_SOURCE_DIR}/Methods/*" "${CMAKE_CURRENT_SOURCE_DIR}/Training")
if(NOT NOORRAY_GAUSSIAN_TRACING)
    list(FILTER shader_folders EXCLUDE REGEX "/GaussianTracing$")
endif()
foreach(folder IN LISTS shader_folders)
    file(GLOB entries CONFIGURE_DEPENDS "${folder}/Shaders/*.slang")
    if(NOT NOORRAY_GAUSSIAN_TRACING)
        list(FILTER entries EXCLUDE REGEX "/GaussianInstances\\.slang$")
    endif()
    if(NOT entries)
        continue()
    endif()
    cmake_path(GET folder FILENAME name)
    cmake_path(GET folder PARENT_PATH parent)
    cmake_path(GET parent FILENAME group)
    set(target DiffRenderer${group}${name}Shaders)
    noorrhi_add_shaders(${target} ${${name}ShaderOptions} ${entries} INCLUDES "${folder}/Shaders" "${folder}/Shaders/Modules" "${CMAKE_CURRENT_SOURCE_DIR}/Shaders"
                        ${${name}ShaderIncludes})
    target_link_libraries(DiffRenderer PRIVATE ${target})
    target_include_directories(DiffRenderer PRIVATE "${folder}/Shaders")
endforeach()
