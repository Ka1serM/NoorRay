cmake_minimum_required(VERSION 3.29)

if(CMAKE_HOST_WIN32)
    set(host_platform Windows)
else()
    set(host_platform Linux)
endif()
set(PLATFORM "${host_platform}" CACHE STRING "The platform of the build: Windows, Linux, or Web")
set_property(CACHE PLATFORM PROPERTY STRINGS Windows Linux Web)
if(NOT PLATFORM MATCHES "^(Windows|Linux|Web)$")
    message(FATAL_ERROR "PLATFORM is '${PLATFORM}'. It must be Windows, Linux, or Web.")
endif()
foreach(platform Windows Linux Web)
    string(TOUPPER "${platform}" platform_name)
    if(PLATFORM STREQUAL platform)
        set(PLATFORM_${platform_name} ON)
    else()
        set(PLATFORM_${platform_name} OFF)
    endif()
endforeach()
#! The web build installs its Emscripten and takes its toolchain file before project().
if(PLATFORM_WEB)
    include(external/RmlKit/cmake/Emsdk.cmake)
endif()

project(NoorRay VERSION 0.1.0 LANGUAGES C CXX)

#! clang-cl is not supported: it has no libc++.
if(NOT CMAKE_C_COMPILER_ID STREQUAL "Clang" OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR MSVC)
    message(FATAL_ERROR "The supported toolchains are Clang on Linux, llvm-mingw on Windows, and Emscripten.")
endif()

option(NOORRAY_BUILD_APP "Build NoorRay.App" ON)
set(NOORRAY_USD_ROOT "" CACHE PATH "An OpenUSD build with include/pxr and lib/. Empty: no USD import.")
if(PLATFORM_WEB)
    set(NOORRAY_USD_ROOT "" CACHE PATH "" FORCE)
endif()

include(external/NoorRHI/cmake/CompilerOptions.cmake)
include(external/NoorRHI/cmake/NoorRHIShaders.cmake)
include(cmake/NoorRayShaders.cmake)
if(NOORRAY_BUILD_APP)
    set(RMLKIT_APP_ID "dev.marcelk.noorray")
    set(RMLKIT_UI_SOURCE_DIR "${PROJECT_SOURCE_DIR}/src/NoorRay.App")
    set(RMLKIT_ASSET_SOURCE_DIR "${PROJECT_SOURCE_DIR}/src/NoorRay.App")
    include(external/RmlKit/cmake/App.cmake)
endif()

function(noorray_code_options target)
    target_compile_options(${target} PRIVATE -fno-exceptions -fno-rtti -Wall -Wextra -Wno-missing-field-initializers)
endfunction()

add_subdirectory(external)
add_subdirectory(src)
